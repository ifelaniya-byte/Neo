class EvolutionaryRotationOrchestrator {
  lineage_loader: LineageLoader;
  
  async spinEvolutionaryLineage(
    input: string,
    family: string,
    include_all_versions: boolean = true
  ): Promise<EvolutionaryRotationResult> {
    // Spin through ALL versions of a family (v0 → v1 → ... → apex)
    
    lineage = this.lineage_loader.lineages.get(family);
    versions_to_process = include_all_versions 
      ? lineage.versions 
      : [lineage.apex];  // Option to test only apex
    
    outputs = [];
    
    for (let version of versions_to_process) {
      // Load version
      model = await this.lineage_loader.loadVersion(version);
      
      // Run inference
      output = await model.predict(input);
      
      // Record with evolution metadata
      evolutionary_output: EvolutionaryOutput = {
        version_id: version.version_id,
        version_number: version.version_number,
        output: output,
        confidence: this.estimateConfidence(output),
        performance_metrics: version.performance_metrics,
        latency_ms: timer.elapsed(),
        position_in_evolution: lineage.versions.indexOf(version) / lineage.versions.length
      };
      
      outputs.push(evolutionary_output);
    }
    
    // Analyze evolution
    evolution_analysis = this.analyzeEvolution(outputs, lineage);
    
    return {
      outputs: outputs,
      lineage: lineage,
      evolution_analysis: evolution_analysis
    };
  }
  
  async spinMultipleFamilies(
    input: string,
    families: string[],
    include_all_versions: boolean = true
  ): Promise<MultiLineageResult> {
    // Spin through multiple families (Qwen + Llama + Phi all at once)
    // This is the COMBO MOVE: all lineages together
    
    results = [];
    
    for (let family of families) {
      result = await this.spinEvolutionaryLineage(input, family, include_all_versions);
      results.push(result);
    }
    
    // Cross-family analysis: do all families improve in same way?
    cross_family_patterns = this.findCommonEvolutionPatterns(results);
    
    return {
      by_family: results,
      cross_family_patterns: cross_family_patterns
    };
  }
  
  analyzeEvolution(
    outputs: EvolutionaryOutput[],
    lineage: ModelLineage
  ): EvolutionAnalysis {
    // Analyze: how did the model improve across versions?
    
    analysis: EvolutionAnalysis = {
      total_versions: outputs.length,
      improvement_trajectory: [],
      breakthrough_points: [],
      stagnation_periods: [],
      predicted_next_iteration: null
    };
    
    // Calculate improvement per version
    for (let i = 0; i < outputs.length - 1; i++) {
      current = outputs[i];
      next = outputs[i + 1];
      
      improvement = this.computeImprovement(current, next);
      
      analysis.improvement_trajectory.push({
        from: current.version_id,
        to: next.version_id,
        improvement_score: improvement,
        delta_type: lineage.evolution_deltas[i]?.delta_type,
        description: lineage.evolution_deltas[i]?.description
      });
      
      // Identify breakthrough (improvement > threshold)
      if (improvement > 0.15) {
        analysis.breakthrough_points.push({
          at_version: next.version_id,
          improvement: improvement,
          likely_cause: lineage.evolution_deltas[i]?.description
        });
      }
      
      // Identify stagnation (improvement < threshold)
      if (improvement < 0.02) {
        analysis.stagnation_periods.push({
          from: current.version_id,
          to: next.version_id,
          stagnation_level: improvement
        });
      }
    }
    
    // Predict next iteration
    analysis.predicted_next_iteration = this.predictNextIteration(outputs, lineage);
    
    return analysis;
  }
  
  predictNextIteration(
    outputs: EvolutionaryOutput[],
    lineage: ModelLineage
  ): PredictedIteration {
    // Learn from patterns: what should next version do?
    
    // Extract successful delta patterns
    successful_patterns = lineage.evolution_deltas.filter(d => d.estimated_importance > 0.7);
    
    // Find patterns that repeat (worked multiple times)
    repeated_patterns = successful_patterns.groupBy(p => p.delta_type)
      .filter(group => group.length > 1);
    
    // Predict: "Next version should focus on X"
    return {
      recommended_focus: repeated_patterns[0]?.delta_type || "training_data",
      expected_improvement: 0.08,  // Estimate based on historical patterns
      confidence: 0.7,
      reasoning: "Based on successful deltas, this type of change tends to work"
    };
  }
  
  findCommonEvolutionPatterns(
    results: MultiLineageResult[]
  ): EvolutionPattern[] {
    // Cross-family analysis: do Qwen, Llama, Phi all improve the same way?
    // If all families improved with "architecture" change, that pattern is strong
    
    patterns = [];
    
    for (let family_result of results) {
      for (let delta of family_result.lineage.evolution_deltas) {
        patterns.push({
          family: family_result.lineage.family,
          delta_type: delta.delta_type,
          improvement: delta.performance_delta
        });
      }
    }
    
    // Group by delta_type
    grouped = patterns.groupBy(p => p.delta_type);
    
    // Return: which delta types are consistent across families?
    return grouped
      .map(group => ({
        delta_type: group[0].delta_type,
        families_affected: group.map(p => p.family),
        avg_improvement: average(group.map(p => p.improvement)),
        consistency: group.length / results.length  // % of families showing this pattern
      }))
      .sort((a, b) => b.consistency - a.consistency);
  }
}

class ApexThresholdGate {
  lineage_loader: LineageLoader;
  marker_system: MarkerRecorder;
  
  async evaluateNewVariant(
    new_variant: VariantOutput,
    test_input: string,
    families: string[] = ["Qwen", "Llama", "Phi"]  // All major families
  ): Promise<ThresholdEvaluation> {
    // Can this new variant enter the rotation?
    // It MUST beat the apex of at least one family
    
    evaluation: ThresholdEvaluation = {
      variant_id: new_variant.variant_id,
      passes_threshold: false,
      apex_comparisons: [],
      recommendation: "REJECT"
    };
    
    for (let family of families) {
      apex = this.lineage_loader.getApex(family);
      
      // Run apex model on same test input
      apex_output = await apex.model.predict(test_input);
      
      // Compare outputs
      comparison = this.compareOutputs(new_variant.output, apex_output);
      
      // Score: does new variant beat apex?
      beats_apex = comparison.similarity < 0.3 || comparison.reasoning_quality > apex_output.reasoning_quality;
      
      evaluation.apex_comparisons.push({
        family: family,
        apex_version: apex.version_id,
        new_variant_score: this.scoreOutput(new_variant.output),
        apex_score: this.scoreOutput(apex_output),
        beats_apex: beats_apex,
        improvement_delta: this.scoreOutput(new_variant.output) - this.scoreOutput(apex_output)
      });
    }
    
    // Verdict: does variant beat ANY apex?
    beats_any_apex = evaluation.apex_comparisons.some(c => c.beats_apex);
    
    if (beats_any_apex) {
      evaluation.passes_threshold = true;
      evaluation.recommendation = "PROMOTE_TO_ROTATION";
      
      // Find which family it beat
      winning_family = evaluation.apex_comparisons.find(c => c.beats_apex).family;
      evaluation.recommendation += ` (Beats ${winning_family} apex)`;
    } else {
      evaluation.passes_threshold = false;
      evaluation.recommendation = "REJECT (Does not beat any apex)";
      
      // Suggest improvements
      weakest_gap = evaluation.apex_comparisons
        .minBy(c => c.improvement_delta);
      
      evaluation.improvement_suggestion = 
        `New variant is ${Math.abs(weakest_gap.improvement_delta) * 100}% weaker than ${weakest_gap.family} apex. Focus on ${weakest_gap.family}-like improvements.`;
    }
    
    return evaluation;
  }
  
  async shouldIntroduceNewVariantNow(
    consensus: ConsensusResult
  ): Promise<boolean> {
    // COMBO TIMING: Only introduce new variants when conditions are perfect
    // = when consensus is strong and system is confident
    
    conditions = {
      agreement_strength_high: consensus.agreement_strength > 0.8,
      confidence_high: consensus.confidence_distribution.mean > 0.85,
      no_recent_failures: await this.checkRecentFailureRate() < 0.05,
      ensemble_stable: await this.checkEnsembleVariance() < 0.1
    };
    
    all_conditions_met = Object.values(conditions).every(c => c === true);
    
    if (all_conditions_met) {
      return true;  // NOW IS THE TIME: Combo move timing
    } else {
      failing_conditions = Object.entries(conditions)
        .filter(([k, v]) => v === false)
        .map(([k, v]) => k);
      
      console.log(`Cannot introduce new variant yet. Failing conditions: ${failing_conditions.join(", ")}`);
      return false;
    }
  }
  
  async checkRecentFailureRate(): Promise<float> {
    // Check: have recent variants been failing?
    recent_evals = await this.marker_system.getRecentEvaluations(days=7);
    failures = recent_evals.filter(e => !e.passes_threshold).length;
    return failures / recent_evals.length;
  }
  
  async checkEnsembleVariance(): Promise<float> {
    // Check: is the ensemble stable?
    recent_consensuses = await this.marker_system.getRecentConsensuses(count=50);
    agreement_scores = recent_consensuses.map(c => c.agreement_strength);
    variance = stddev(agreement_scores);
    return variance;
  }
}

class EvolutionaryLearningEngine {

  async extractEvolutionaryLessons(): Promise<EvolutionaryInsights> {
    // Learn: what patterns made models better?
    
    insights: EvolutionaryInsights = {
      delta_patterns: [],
      breakthrough_patterns: [],
      stagnation_patterns: [],
      meta_model: null,
      confidence: 0.0
    };
    
    // Process each family
    for (let lineage of this.lineage_loader.lineages.values()) {
      // Analyze each delta
      for (let delta of lineage.evolution_deltas) {
        pattern = {
          family: lineage.family,
          delta_type: delta.delta_type,
          description: delta.description,
          performance_improvement: delta.performance_delta,
          importance: delta.estimated_importance,
          success: delta.performance_delta > 0.05
        };
        
        insights.delta_patterns.push(pattern);
        
        if (delta.performance_delta > 0.15) {
          insights.breakthrough_patterns.push(pattern);
        }
        
        if (delta.performance_delta < 0.02) {
          insights.stagnation_patterns.push(pattern);
        }
      }
    }
    
    // Meta-learning: what's the pattern?
    insights.meta_model = this.buildMetaModel(insights.delta_patterns);
    
    return insights;
  }
  
  buildMetaModel(patterns: DeltaPattern[]): MetaModel {
    // Learn: what sequence of delta types leads to best outcomes?
    
    // Most common successful delta type
    successful_patterns = patterns.filter(p => p.success);
    delta_type_frequency = successful_patterns.groupBy(p => p.delta_type);
    most_common_successful = delta_type_frequency
      .maxBy(group => group.length)[0];
    
    return {
      most_effective_delta_type: most_common_successful.delta_type,
      typical_improvement_per_iteration: average(successful_patterns.map(p => p.performance_improvement)),
      success_rate_by_type: {
        // For each delta type, what % succeed?
        "architecture": successful_patterns.filter(p => p.delta_type === "architecture").length / 
                        patterns.filter(p => p.delta_type === "architecture").length,
        "training_data": successful_patterns.filter(p => p.delta_type === "training_data").length /
                         patterns.filter(p => p.delta_type === "training_data").length,
        "fine_tune": ...,
        "quantization": ...
      },
      predicted_next_improvement: 0.08,  // Based on pattern
      confidence: 0.75
    };
  }
  
  async recommendNewVariantFocus(): Promise<string> {
    // Given what we learned, what should new variants focus on?
    
    insights = await this.extractEvolutionaryLessons();
    
    // Find most effective delta type with lowest recent usage
    most_effective_type = insights.meta_model.most_effective_delta_type;
    recent_usage = await this.countRecentDeltaType(most_effective_type, days=30);
    
    if (recent_usage < 3) {
      // We haven't tried this recently, but it worked before
      return `Apply ${most_effective_type} changes (historically successful pattern)`;
    }
    
    // Otherwise recommend next promising type
    second_most_effective = insights.breakthrough_patterns
      .groupBy(p => p.delta_type)
      .sort((a, b) => b.length - a.length)[1][0].delta_type;
    
    return `Try ${second_most_effective} changes (proven effective across families)`;
  }

class EvolutionaryRotationConsensusSystem:
    def __init__(self, lineage_manifest, main_llm_model):
        self.loader = LineageLoader(lineage_manifest)
        self.rotator = EvolutionaryRotationOrchestrator(self.loader)
        self.marker = MarkerRecorder()
        self.consensus = ConsensusAggregator()
        self.analyzer = EvolutionAnalyzer()
        self.gate = ApexThresholdGate()
        self.main_llm = main_llm_model
    
    async def process_query(self, input_text, families=["Qwen", "Llama", "Phi"]):
        # STEP 1: Spin all lineages
        rotation_result = await self.rotator.spin_all_families(input_text, families)
        
        # STEP 2: Record markers
        for family in families:
            for output in rotation_result[family]:
                self.marker.record_variant_output(
                    input_text, output.output, output.version_id, 
                    output.position_in_evolution
                )
        
        # STEP 3: Extract evolution
        for family in families:
            analysis = self.analyzer.extract_patterns(
                rotation_result[family], self.loader.lineages[family]
            )
            print(f"{family}: {analysis.breakthrough_points} breakthroughs")
        
        # STEP 4: Aggregate consensus (apex only)
        apex_outputs = {
            family: rotation_result[family][-1]  # Last = apex
            for family in families
        }
        consensus = self.consensus.aggregate(apex_outputs)
        
        # STEP 5: Check combo timing
        can_introduce = self.gate.should_introduce_now(consensus)
        
        # STEP 6: Main LLM synthesis
        synthesis_prompt = self.build_evolutionary_prompt(
            input_text, rotation_result, consensus
        )
        main_output = await self.main_llm.generate(synthesis_prompt)
        
        # STEP 7: Extract lessons
        lessons = self.analyzer.build_meta_model(
            [self.analyzer.extract_patterns(rotation_result[f], 
             self.loader.lineages[f]) for f in families]
        )
        
        return {
            "answer": main_output,
            "confidence": consensus.agreement_strength,
            "evolution_insights": lessons,

I couldn't find "State Resolution Operation" as a standardized, universally-recognized acronym in computing — "SRO" turns out to be heavily overloaded (nuclear reactor operator, self-regulatory organization, service order systems, "sharable read-only" in some OS contexts). The closest formal match is from a patent framework called PERCos, where SRO stands for "Specification, Resolution, and Operational" processing — not quite what you wrote, and a fairly obscure/niche system, so probably not what you're after.

That said, "state resolution" as a general computing concept is real and shows up in a few distinct places. Here's the landscape:

1. Distributed systems — resolving conflicting state
When multiple nodes/replicas have diverging views of "the truth," a state resolution algorithm decides what the canonical state should be. The clearest named example is the Matrix protocol's State Resolution Algorithm, which merges conflicting room state events from federated servers into one agreed state.

2. Control loops / reconciliation (Kubernetes-style)
A controller continuously compares desired state vs actual state and takes actions to converge them. This is often called reconciliation rather than "resolution," but it's the same underlying operation: resolve differences → produce a consistent state.

3. Consensus algorithms
In systems like Raft or Paxos, "resolving state" means getting all nodes to agree on a single value/log entry despite failures or network partitions — directly relevant to something like a rotating multi-model ensemble, where you need a rule for reconciling disagreeing model outputs into one final state.

4. Finite state machines / event sourcing
Resolving state = computing the current state by replaying or folding a sequence of events/transitions, especially when transitions can be ambiguous or conflicting.

5. Game netcode
Client-side prediction + server reconciliation: the client guesses state, the server periodically sends authoritative state, and the client "resolves" the difference (rewinding and replaying).

class JobState(Enum):
    """File processing states."""
    INGESTED = "INGESTED"
    VALIDATING = "VALIDATING"
    VALIDATED = "VALIDATED"
    COMPRESSING = "COMPRESSING"
    COMPRESSED = "COMPRESSED"
    CHUNKING = "CHUNKING"
    CHUNKED = "CHUNKED"
    UPLOADING = "UPLOADING"
    UPLOADED = "UPLOADED"
    FINALIZING = "FINALIZING"
    COMPLETED = "COMPLETED"
    FAILED = "FAILED"
 
@dataclass
class JobRecord:
    """Job state tracked in database."""
    file_id: str
    filename: str
    original_path: str
    state: JobState = JobState.INGESTED
    agent_id: Optional[str] = None
    media_id: Optional[str] = None  # Twitter media_id
    error_message: Optional[str] = None
    retries: int = 0
    compressed_path: Optional[str] = None
    chunk_manifest: Dict = field(default_factory=dict)  # {segment_index: path}
    created_at: str = field(default_factory=lambda: datetime.now().isoformat())
    updated_at: str = field(default_factory=lambda: datetime.now().isoformat())
    file_hash: Optional[str] = None
    original_size: int = 0
    compressed_size: int = 0
 
class JobDatabase:
    """SQLite-backed job state machine."""
    
    def __init__(self, db_path: Path):
        self.db_path = db_path
        self._init_schema()
    
    def _init_schema(self):
        """Initialize database schema."""
        with sqlite3.connect(self.db_path) as conn:
            conn.execute("""
                CREATE TABLE IF NOT EXISTS jobs (
                    file_id TEXT PRIMARY KEY,
                    filename TEXT NOT NULL,
                    original_path TEXT NOT NULL,
                    state TEXT DEFAULT 'INGESTED',
                    agent_id TEXT,
                    media_id TEXT,
                    error_message TEXT,
                    retries INTEGER DEFAULT 0,
                    compressed_path TEXT,
                    chunk_manifest TEXT,  -- JSON
                    created_at TEXT,
                    updated_at TEXT,
                    file_hash TEXT,
                    original_size INTEGER DEFAULT 0,
                    compressed_size INTEGER DEFAULT 0
                )
            """)
            
            conn.execute("""
                CREATE TABLE IF NOT EXISTS segments (
                    media_id TEXT NOT NULL,
                    segment_index INTEGER NOT NULL,
                    size INTEGER,
                    status TEXT DEFAULT 'pending',  -- pending, uploaded, failed
                    PRIMARY KEY (media_id, segment_index)
                )
            """)
            
            conn.commit()
    
    def create_job(self, filepath: str) -> str:
        """Create a new job record."""
        file_id = hashlib.md5(filepath.encode()).hexdigest()
        original_size = os.path.getsize(filepath)
        
        record = JobRecord(
            file_id=file_id,
            filename=Path(filepath).name,
            original_path=filepath,
            original_size=original_size
        )
        
        with sqlite3.connect(self.db_path) as conn:
            conn.execute("""
                INSERT INTO jobs 
                (file_id, filename, original_path, state, created_at, updated_at, original_size)
                VALUES (?, ?, ?, ?, ?, ?, ?)
            """, (
                record.file_id, record.filename, record.original_path,
                record.state.value, record.created_at, record.updated_at,
                record.original_size
            ))
            conn.commit()
        
        logger.info(f"Created job {file_id} for {record.filename}")
        return file_id
    
    def get_job(self, file_id: str) -> Optional[JobRecord]:
        """Retrieve a job record."""
        with sqlite3.connect(self.db_path) as conn:
            cursor = conn.execute("SELECT * FROM jobs WHERE file_id = ?", (file_id,))
            row = cursor.fetchone()
        
        if not row:
            return None
        
        return self._row_to_record(row)
    
    def get_jobs_in_state(self, state: JobState, limit: int = 10) -> List[JobRecord]:
        """Get all jobs in a specific state."""
        with sqlite3.connect(self.db_path) as conn:
            cursor = conn.execute(
                "SELECT * FROM jobs WHERE state = ? AND agent_id IS NULL LIMIT ?",
                (state.value, limit)
            )
            rows = cursor.fetchall()
        
        return [self._row_to_record(row) for row in rows]
    
    def claim_job(self, file_id: str, agent_id: str) -> bool:
        """Atomically claim a job (for crash recovery)."""
        with sqlite3.connect(self.db_path) as conn:
            cursor = conn.execute(
                "UPDATE jobs SET agent_id = ? WHERE file_id = ? AND agent_id IS NULL",
                (agent_id, file_id)
            )
            conn.commit()
            return cursor.rowcount > 0
    
    def update_job_state(
        self, 
        file_id: str, 
        new_state: JobState, 
        agent_id: Optional[str] = None,
        **kwargs
    ) -> bool:
        """Update job state atomically."""
        updates = {"state": new_state.value, "updated_at": datetime.now().isoformat()}
        updates.update(kwargs)
        
        placeholders = ", ".join([f"{k} = ?" for k in updates.keys()])
        values = list(updates.values()) + [file_id]
        
        with sqlite3.connect(self.db_path) as conn:
            conn.execute(
                f"UPDATE jobs SET {placeholders} WHERE file_id = ?",
                values
            )
            conn.commit()
        
        return True
    
    def record_segment_upload(self, media_id: str, segment_index: int, size: int, status: str):
        """Track uploaded segments for resumability."""
        with sqlite3.connect(self.db_path) as conn:
            conn.execute(

                (media_id, segment_index, size, status)
            )
            conn.commit()
    
    def get_uploaded_segments(self, media_id: str) -> List[int]:

class FileStabilityMonitor:
    """Detect when a file has finished writing."""
    
    @staticmethod
    def wait_until_stable(
        filepath: str,
        poll_interval: int = Config.FILE_STABILITY_POLL_INTERVAL,
        stable_checks: int = Config.FILE_STABILITY_CHECKS,
        timeout: int = 300
    ) -> bool:
        """
        Block until file size stops changing.
        
        Returns True if stable, False if timeout.
        """
        start_time = time.time()
        prev_size = -1
        checks = 0
        
        while time.time() - start_time < timeout:
            try:
                curr_size = os.path.getsize(filepath)
            except OSError:
                logger.warning(f"File disappeared during stability check: {filepath}")
                return False
            
            if curr_size == prev_size and curr_size > 0:
                checks += 1
                if checks >= stable_checks:
                    logger.info(f"File stable after {time.time() - start_time:.1f}s: {filepath}")
                    return True
            else:
                checks = 0
                prev_size = curr_size
            
            time.sleep(poll_interval)
        
        logger.error(f"File stability timeout after {timeout}s: {filepath}")
        return False
    
    @staticmethod
    def compute_file_hash(filepath: str, algorithm: str = 'md5', chunk_size: int = 1024*1024) -> str:
        """Compute hash of large file efficiently."""
        hasher = hashlib.new(algorithm)
        with open(filepath, 'rb') as f:
            while chunk := f.read(chunk_size):
                hasher.update(chunk)
        return hasher.hexdigest()

class ValidatorAgent:
    """Check file integrity and compatibility."""
    
    def __init__(self, db: JobDatabase):
        self.db = db
        self.agent_id = "validator_001"
    
    def process(self, job: JobRecord) -> bool:
        """Validate file format and size."""
        try:
            self.db.update_job_state(job.file_id, JobState.VALIDATING, self.agent_id)
            
            # Check file exists and is readable
            if not os.path.exists(job.original_path):
                raise FileNotFoundError(f"File not found: {job.original_path}")
            
            # Compute hash for integrity checking
            logger.info(f"Computing hash for {job.filename}")
            file_hash = FileStabilityMonitor.compute_file_hash(job.original_path)
            
            # Validate format (basic check)
            if job.filename.lower().endswith(('.mp4', '.mov', '.avi', '.mkv', '.zip', '.tar')):
                logger.info(f"Valid format: {job.filename}")
            else:
                raise ValueError(f"Unsupported format: {job.filename}")
            
            # Update database
            self.db.update_job_state(
                job.file_id,
                JobState.VALIDATED,
                file_hash=file_hash
            )
            
            logger.info(f"Validation passed: {job.filename} (hash: {file_hash})")
            return True
            
        except Exception as e:
            logger.error(f"Validation failed for {job.file_id}: {str(e)}")
            self.db.update_job_state(
                job.file_id,
                JobState.FAILED,
                error_message=str(e)
            )
            return False

class CompressorAgent:

        """Compress file using ffmpeg."""
        with self.semaphore:  # Enforce concurrency limit
            try:
                self.db.update_job_state(job.file_id, JobState.COMPRESSING, self.agent_id)
                
                # Skip compression if already small enough
                if job.original_size < Config.TWITTER_MAX_FILE_SIZE:
                    logger.info(f"File already under size limit, skipping compression: {job.filename}")
                    self.db.update_job_state(
                        job.file_id,
                        JobState.COMPRESSED,
                        compressed_path=job.original_path,
                        compressed_size=job.original_size
                    )
                    return True
                
                # Determine output format
                input_path = job.original_path
                output_path = Config.TEMP_DIR / f"{job.file_id}_compressed.mp4"
                
                # FFmpeg command for video compression
                cmd = [
                    'ffmpeg',
                    '-i', str(input_path),
                    '-c:v', 'h264',
                    '-preset', Config.COMPRESSION_PRESET,
                    '-b:v', Config.MAX_VIDEO_BITRATE,
                    '-c:a', 'aac',
                    '-b:a', '128k',
                    '-y',  # Overwrite output
                    str(output_path)
                ]
                
                logger.info(f"Starting compression: {job.filename}")
                logger.debug(f"FFmpeg command: {' '.join(cmd)}")
                
                # Run ffmpeg
                result = subprocess.run(
                    cmd,
                    capture_output=True,
                    text=True,
                    timeout=3600  # 1 hour timeout
                )
                
                if result.returncode != 0:
                    raise RuntimeError(f"FFmpeg failed: {result.stderr}")
                
                # Verify output
                if not output_path.exists():
                    raise RuntimeError(f"Output file not created: {output_path}")
                
                compressed_size = os.path.getsize(output_path)
                reduction = (1 - compressed_size / job.original_size) * 100
                
                logger.info(
                    f"Compression complete: {job.filename} "
                    f"({job.original_size/1e9:.2f}GB → {compressed_size/1e9:.2f}GB, {reduction:.1f}% reduction)"
                )
                
                # Update database
                self.db.update_job_state(
                    job.file_id,
                    JobState.COMPRESSED,
                    compressed_path=str(output_path),
                    compressed_size=compressed_size
                )
                
                return True
                
            except Exception as e:
                logger.error(f"Compression failed for {job.file_id}: {str(e)}")
                self.db.update_job_state(
                    job.file_id,
                    JobState.FAILED,
                    error_message=f"Compression error: {str(e)}"
                )
                return False

class ChunkerAgent:
    """Split files into uploadable chunks."""
    
    def __init__(self, db: JobDatabase, chunk_size: int = Config.TWITTER_CHUNK_SIZE):
        self.db = db
        self.agent_id = "chunker_001"
        self.chunk_size = chunk_size
    
    def process(self, job: JobRecord) -> bool:
        """Split file into chunks for upload."""
        try:
            self.db.update_job_state(job.file_id, JobState.CHUNKING, self.agent_id)
            
            # Use compressed path if available, otherwise original
            file_to_chunk = job.compressed_path or job.original_path
            file_size = os.path.getsize(file_to_chunk)
            
            # If file is small, no chunking needed
            if file_size < self.chunk_size:
                logger.info(f"File smaller than chunk size, no chunking needed: {job.filename}")
                self.db.update_job_state(
                    job.file_id,
                    JobState.CHUNKED,
                    chunk_manifest=json.dumps({0: file_to_chunk})
                )
                return True
            
            # Create chunks
            chunk_manifest = {}
            with open(file_to_chunk, 'rb') as f:
                segment_index = 0
                while True:
                    chunk_data = f.read(self.chunk_size)
                    if not chunk_data:
                        break
                    
                    chunk_path = Config.TEMP_DIR / f"{job.file_id}_chunk_{segment_index}"
                    with open(chunk_path, 'wb') as chunk_file:
                        chunk_file.write(chunk_data)
                    
                    chunk_manifest[str(segment_index)] = str(chunk_path)
                    logger.debug(f"Created chunk {segment_index}: {len(chunk_data)} bytes")
                    segment_index += 1
            
            logger.info(f"Chunking complete: {len(chunk_manifest)} chunks for {job.filename}")
            
            # Update database
            self.db.update_job_state(
                job.file_id,
                JobState.CHUNKED,
                chunk_manifest=json.dumps(chunk_manifest)
            )
            
            return True
            
        except Exception as e:
            logger.error(f"Chunking failed for {job.file_id}: {str(e)}")
            self.db.update_job_state(
                job.file_id,
                JobState.FAILED,
                error_message=f"Chunking error: {str(e)}"
            )
            return False

class TwitterUploader:
    """Upload to Twitter using chunked media upload API."""
    
    def __init__(
        self,
        api_key: str,
        api_secret: str,
        access_token: str,
        access_secret: str,
        db: JobDatabase
    ):
        self.db = db
        self.agent_id = "twitter_uploader_001"
        self.api_key = api_key
        self.api_secret = api_secret
        self.access_token = access_token
        self.access_secret = access_secret
        
        # OAuth headers
        self.headers = {
            "Authorization": f"Bearer {access_token}"
        }
    
    def init_upload(self, total_bytes: int, media_type: str = "video/mp4") -> Optional[str]:
        """Initialize chunked upload (INIT command)."""
        url = "https://upload.twitter.com/1.1/media/upload.json"
        
        params = {
            "command": "INIT",
            "total_bytes": total_bytes,
            "media_type": media_type
        }
        
        try:
            response = requests.post(url, headers=self.headers, params=params, timeout=30)
            response.raise_for_status()
            data = response.json()
            media_id = data.get("media_id_string") or str(data.get("media_id"))
            logger.info(f"Initialized upload with media_id: {media_id}")
            return media_id
        except Exception as e:
            logger.error(f"Upload INIT failed: {str(e)}")
            return None
    
    def append_chunk(self, media_id: str, segment_index: int, chunk_data: bytes) -> bool:
        """Append chunk to upload (APPEND command)."""
        url = "https://upload.twitter.com/1.1/media/upload.json"
        
        params = {
            "command": "APPEND",
            "media_id": media_id,
            "segment_index": segment_index
        }
        
        try:
            response = requests.post(
                url,
                headers=self.headers,
                params=params,
                data=chunk_data,
                timeout=Config.TWITTER_CHUNK_UPLOAD_TIMEOUT
            )
            response.raise_for_status()
            logger.debug(f"Appended segment {segment_index} ({len(chunk_data)} bytes)")
            return True
        except requests.exceptions.HTTPError as e:
            if e.response.status_code == 429:  # Rate limit
                logger.warning(f"Rate limited on segment {segment_index}, will retry")
                return False
            else:
                logger.error(f"Upload APPEND failed: {str(e)}")
                return False
        except Exception as e:
            logger.error(f"Upload APPEND failed: {str(e)}")
            return False
    
    def finalize_upload(self, media_id: str) -> bool:
        """Finalize upload (FINALIZE command)."""
        url = "https://upload.twitter.com/1.1/media/upload.json"
        
        params = {
            "command": "FINALIZE",
            "media_id": media_id
        }
        
        try:
            response = requests.post(url, headers=self.headers, params=params, timeout=60)
            response.raise_for_status()
            logger.info(f"Finalized upload for media_id: {media_id}")
            return True
        except Exception as e:
            logger.error(f"Upload FINALIZE failed: {str(e)}")
            return False
    
    def check_status(self, media_id: str) -> Optional[Dict]:
        """Check upload status (STATUS command)."""
        url = "https://upload.twitter.com/1.1/media/upload.json"
        
        params = {
            "command": "STATUS",
            "media_id": media_id
        }
        
        try:
            response = requests.get(url, headers=self.headers, params=params, timeout=30)
            response.raise_for_status()
            status = response.json()
            state = status.get("media_info", {}).get("processing_info", {}).get("state")
            logger.info(f"Media {media_id} status: {state}")
            return status
        except Exception as e:
            logger.error(f"Status check failed: {str(e)}")
            return None
 
class TwitterUploaderAgent:
    """Agent wrapper for Twitter uploads with retry logic."""
    
    def __init__(
        self,
        api_key: str,
        api_secret: str,
        access_token: str,
        access_secret: str,
        db: JobDatabase,
        max_concurrent: int = Config.MAX_CONCURRENT_UPLOADS
    ):
        self.uploader = TwitterUploader(api_key, api_secret, access_token, access_secret, db)
        self.db = db
        self.agent_id = "twitter_uploader_001"
        self.semaphore = threading.Semaphore(max_concurrent)
    
    def process(self, job: JobRecord) -> bool:
        """Upload file to Twitter using chunked API with resumability."""
        with self.semaphore:
            try:
                self.db.update_job_state(job.file_id, JobState.UPLOADING, self.agent_id)
                
                # Load chunk manifest
                if not job.chunk_manifest:
                    logger.error(f"No chunks available for {job.file_id}")
                    raise ValueError("Job not chunked")
                
                chunks = {int(k): v for k, v in job.chunk_manifest.items()}
                
                # Initialize upload
                total_size = sum(os.path.getsize(chunks[i]) for i in sorted(chunks.keys()))
                media_id = self.uploader.init_upload(total_size, media_type="video/mp4")
                
                if not media_id:
                    raise RuntimeError("Failed to initialize upload")
                
                # Upload chunks with resumability
                uploaded_segments = self.db.get_uploaded_segments(media_id)
                logger.info(f"Resuming upload, already uploaded segments: {uploaded_segments}")
                
                for segment_index in sorted(chunks.keys()):
                    # Skip already uploaded segments
                    if segment_index in uploaded_segments:
                        logger.info(f"Segment {segment_index} already uploaded, skipping")
                        continue
                    
                    chunk_path = chunks[segment_index]
                    with open(chunk_path, 'rb') as f:
                        chunk_data = f.read()
                    
                    # Upload with retry
                    for attempt in range(Config.MAX_RETRIES):
                        if self.uploader.append_chunk(media_id, segment_index, chunk_data):
                            self.db.record_segment_upload(media_id, segment_index, len(chunk_data), "uploaded")
                            break
                        else:
                            if attempt < Config.MAX_RETRIES - 1:
                                delay = Config.RETRY_INITIAL_DELAY * (Config.RETRY_BACKOFF_FACTOR ** attempt)
                                logger.warning(f"Segment {segment_index} upload failed, retrying in {delay}s")
                                time.sleep(delay)
                            else:
                                raise RuntimeError(f"Failed to upload segment {segment_index} after {Config.MAX_RETRIES} attempts")
                    
                    # Rate limiting
                    time.sleep(Config.TWITTER_RATE_LIMIT_DELAY)
                
                # Finalize upload
                if not self.uploader.finalize_upload(media_id):
                    raise RuntimeError("Failed to finalize upload")
                
                # Poll for processing completion
                logger.info(f"Waiting for Twitter to process media {media_id}")
                start_time = time.time()
                while time.time() - start_time < 600:  # 10 minute timeout
                    status = self.uploader.check_status(media_id)
                    if status:
                        state = status.get("media_info", {}).get("processing_info", {}).get("state")
                        if state == "succeeded":
                            logger.info(f"Upload succeeded: {media_id}")
                            break
                        elif state == "failed":
                            raise RuntimeError(f"Media processing failed: {status}")
                    
                    time.sleep(5)  # Poll every 5 seconds
                
                # Update database
                self.db.update_job_state(
                    job.file_id,
                    JobState.UPLOADED,
                    media_id=media_id
                )
                
                logger.info(f"Upload complete for {job.filename}: {media_id}")
                return True
                
            except Exception as e:
                logger.error(f"Upload failed for {job.file_id}: {str(e)}")
                
                # Retry logic
                if job.retries < Config.MAX_RETRIES:
                    delay = Config.RETRY_INITIAL_DELAY * (Config.RETRY_BACKOFF_FACTOR ** job.retries)
                    logger.info(f"Will retry {job.file_id} in {delay}s (attempt {job.retries + 1}/{Config.MAX_RETRIES})")
                    time.sleep(delay)
                    self.db.update_job_state(
                        job.file_id,
                        JobState.CHUNKED,  # Revert to chunked state
                        retries=job.retries + 1
                    )
                else:
                    self.db.update_job_state(
                        job.file_id,
                        JobState.FAILED,
                        error_message=f"Upload failed after {Config.MAX_RETRIES} attempts: {str(e)}"
                    )
                
                return False

class WorkspaceWatcher(FileSystemEventHandler):
    """Watch shared folder for new files."""
    
    def __init__(self, db: JobDatabase):
        self.db = db
    
    def on_created(self, event):
        """File created event."""
        if event.is_directory or event.src_path.endswith(('.tmp', '.lock', '.part')):
            return
        
        filepath = event.src_path
        logger.info(f"File detected: {filepath}")
        
        # Wait for file stability
        if not FileStabilityMonitor.wait_until_stable(filepath):
            logger.warning(f"File stability check failed: {filepath}")
            return
        
        # Create job
        file_id = self.db.create_job(filepath)
        logger.info(f"Job created: {file_id}")
 
class AgentCoordinator:
    """Orchestrate agents based on job state."""
    
    def __init__(
        self,
        db: JobDatabase,
        twitter_config: Optional[Dict] = None,
        worker_pool_size: int = Config.WORKER_POOL_SIZE
    ):
        self.db = db
        self.validator = ValidatorAgent(db)
        self.compressor = CompressorAgent(db)
        self.chunker = ChunkerAgent(db)
        
        if twitter_config:
            self.twitter_uploader = TwitterUploaderAgent(
                api_key=twitter_config['api_key'],
                api_secret=twitter_config['api_secret'],
                access_token=twitter_config['access_token'],
                access_secret=twitter_config['access_secret'],
                db=db
            )
        else:
            self.twitter_uploader = None
        
        self.executor = ThreadPoolExecutor(max_workers=worker_pool_size)
        self.running = False
    
    def start(self):
        """Start coordination loop."""
        self.running = True
        
        # File watcher
        observer = Observer()
        observer.schedule(WorkspaceWatcher(self.db), path=str(Config.WORKSPACE_DIR), recursive=False)
        observer.start()
        
        logger.info("🚀 Agent OS Started - Watching for files...")
        
        # Main coordination loop
        try:
            while self.running:
                self._process_queue()
                time.sleep(5)  # Check queue every 5 seconds
        except KeyboardInterrupt:
            logger.info("Shutting down...")
            observer.stop()
            observer.join()
            self.executor.shutdown(wait=True)
    
    def _process_queue(self):
        """Process jobs in each state."""
        # Process jobs by state in order
        states_to_process = [
            (JobState.INGESTED, self.validator),
            (JobState.VALIDATED, self.compressor),
            (JobState.COMPRESSED, self.chunker),
            (JobState.CHUNKED, self.twitter_uploader),
        ]
        
        for state, agent in states_to_process:
            if agent is None:
                continue
            
            jobs = self.db.get_jobs_in_state(state, limit=5)
            
            for job in jobs:
                # Claim job (atomic)
                if not self.db.claim_job(job.file_id, agent.agent_id):
                    continue  # Already claimed by another worker
                
                # Submit to executor
                self.executor.submit(agent.process, job)

def main():
    """Main entry point."""
    # Initialize database
    db = JobDatabase(Config.DB_PATH)
    
    # Twitter config (from environment or config file)
    twitter_config = {
        'api_key': os.getenv('TWITTER_API_KEY'),
        'api_secret': os.getenv('TWITTER_API_SECRET'),
        'access_token': os.getenv('TWITTER_ACCESS_TOKEN'),
        'access_secret': os.getenv('TWITTER_ACCESS_SECRET'),
    }

    coordinator = AgentCoordinator(
        db,
        twitter_config=twitter_config if all(twitter_config.values()) else None
    )
    
    # Start
    coordinator.start()
